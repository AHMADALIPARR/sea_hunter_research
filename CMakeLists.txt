cmake_minimum_required(VERSION 3.16)
project(sea_hunter_research_baseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(sim_nav_safety src/sim_nav_safety.cpp)
add_executable(phase_demo src/phase_demo.cpp)

if(MSVC)
  target_compile_options(sim_nav_safety PRIVATE /W4)
  target_compile_options(phase_demo PRIVATE /W4)
else()
  target_compile_options(sim_nav_safety PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(phase_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(BUILD_GO "Build the C++ Go/AlphaZero subtree" ON)
if(BUILD_GO)
  add_subdirectory(cpp)
endif()
